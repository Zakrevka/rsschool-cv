    using namespace std;

    void enter(int* array, int size, ifstream& file) {
        for (int i = 0; i < size; i++) {
            file >> array[i];
        }
    }

    void print(int* array, int size) {
        for (int i = 0; i < size; i++) {
            cout << setw(5) << array[i];
        }   
        cout << endl;
    }   

    void uniun(int* arrayA, int n, int* arrayB, int m) {
        int amount = 0, i = 0, j = 0;
        int* answer = new int[n + m];

        while (i < n && j < m) {
            if (arrayA[i] >= arrayB[j]) {
                answer[amount++] = arrayB[j++];
            } else {
                answer[amount++] = arrayA[i++];
            }
        }

        while (i < n) answer[amount++] = arrayA[i++];
        while (j < m) answer[amount++] = arrayB[j++];

        print(answer, n + m);
        ofstream answer2("answer.txt");
        for (int k = 0; k < n + m; k++) {
            answer2 << answer[k] << ' ';
        }
        delete[] answer;
        answer2.close();
    }

    int main() {

        cout << "Enter n\n"; cin >> n;

        int* arrayA = new int[n];
        int* arrayB = new int[m];

        ifstream file("file.txt");
        if (!file) {
            cerr << "Ошибка: файл не найден!" << endl;
            return 1;
        }

        enter(arrayA, n, file);
        enter(arrayB, m, file);
        print(arrayA, n);
        print(arrayB, m);
        uniun(arrayA, n, arrayB, m);

        file.close();
        delete[] arrayA;
        delete[] arrayB;

        return 0;
    }
